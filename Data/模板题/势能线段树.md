## 势能线段树

```c++
#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
using i64 = long long;

constexpr int N = 1e5 + 10;

struct Info
{
	i64 val;
	int mx;

	Info operator+(const Info &rhs) const
	{
		Info res;
		res.val = val + rhs.val;
		res.mx = max(mx, rhs.mx);
		return res;
	}

	Info &operator+=(const Info &rhs) 
	{
		*this = *this + rhs;
		return *this;
	}
}tr[N << 2];

void pushup(int u)
{
	tr[u] = tr[u << 1] + tr[u << 1 | 1];
}

int n, m;
int a[N];

void build(int u, int l, int r)
{
	if(l == r)
	{
		tr[u].val = tr[u].mx = a[l];
		return ;
	}
	int mid = l + r >> 1;
	build(u << 1, l, mid), build(u << 1 | 1, mid + 1, r);
	pushup(u);
}

i64 rangeSum(int u, int l, int r, int ql, int qr)
{
	if(l >= ql && r <= qr) return tr[u].val;

	int mid = l + r >> 1;

	i64 res = 0;
	if(ql <= mid) res += rangeSum(u << 1, l, mid, ql, qr);
	if(qr >= mid + 1) res += rangeSum(u << 1 | 1, mid + 1, r, ql, qr);
	return res;
}

i64 rangeSum(int ql, int qr)
{
	return rangeSum(1, 1, n, ql, qr);
}

void modify(int u, int l, int r, int pos, int x)
{
	if(l == r)
	{
		tr[u].val = tr[u].mx = x;
		return ;
	}

	int mid = l + r >> 1;
	if(pos <= mid) modify(u << 1, l, mid, pos, x);
	else modify(u << 1 | 1, mid + 1, r, pos, x);
	pushup(u);
}

void modify(int pos, int x)
{
	return modify(1, 1, n, pos, x);
}

void rangeMod(int u, int l, int r, int ql, int qr, int mod)
{
	if(tr[u].mx < mod) return ;

	if(l == r)
	{
		tr[u].val %= mod;
		tr[u].mx %= mod;
		return ;
	}

	int mid = l + r >> 1;
	if(ql <= mid) rangeMod(u << 1, l, mid, ql, qr, mod);
	if(qr >= mid + 1) rangeMod(u << 1 | 1, mid + 1, r, ql, qr, mod);
	pushup(u);
}

void rangeMod(int ql, int qr, int mod)
{
	return rangeMod(1, 1, n, ql, qr, mod);
}

signed main()
{
	ios::sync_with_stdio(false);
	cin.tie(nullptr);

	cin >>n >>m;
	for(int i = 1; i <= n; i ++) cin >>a[i];

	build(1, 1, n);

	while(m --)
	{
		int op; cin >>op;
		if(op == 1)
		{
			int l, r; cin >>l >>r;
			cout <<rangeSum(l, r) <<endl;
		}
		else if(op == 2)
		{
			int l, r, mod; cin >>l >>r >>mod;
			rangeMod(l, r, mod);
		}
		else
		{
			int pos, x; cin >>pos >>x;
			modify(pos, x);
		}
	}
	return 0;
}
```
