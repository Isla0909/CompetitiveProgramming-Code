## 整除分块

```c++
#include <bits/stdc++.h>
#define endl '\n'
using namespace std;
using i64 = long long;

signed main()
{
	ios::sync_with_stdio(false);
	cin.tie(nullptr);

	int T; cin >>T;
	while(T --)
	{
		i64 L, R; cin >>L >>R;
		i64 gap = R - L;

		i64 l = (gap + 1) / 2, ans = l;
		while(l <= gap)
		{
			i64 q1 = R / l, r1 = R / q1;
			i64 q2 = (L - 1) / l, r2;

			if(q2 == 0) r2 = gap;
			else r2 = (L - 1) / q2;

			i64 r = min({r1, r2, gap});
			if(q1 - q2 >= 2) ans = max(ans, r);

			l = r + 1;
		}
		cout <<ans <<endl;
	}
	return 0;
}
```
